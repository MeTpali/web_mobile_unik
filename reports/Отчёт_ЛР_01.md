# 2. Выполнение

## 2.1. Средства разработки и структура решения

Разработан генератор случайных целых чисел с браузерным интерфейсом. Сервер реализован на Node.js и Express, клиент — на React. Vite используется для разработки и сборки HTML, CSS и JavaScript. Рабочая станция — Ubuntu 24.04; проект требует Node.js версии 24 не ниже 24.13.0. Версии библиотек и команды проекта указаны в package.json.

Диапазон ограничен значениями от −1 000 000 000 до 1 000 000 000. Отрицательные и одинаковые границы допустимы. Сервер принимает только целые числа; обе границы включаются в диапазон генерации. Пустой ввод, дробные значения и обратный диапазон сопровождаются сообщением об ошибке.

## 2.2. Взаимодействие клиента и сервера

При отправке формы клиент выполняет асинхронный POST-запрос к относительному адресу /api/random. Сервер проверяет границы, генерирует число и возвращает JSON. React обновляет результат без перезагрузки страницы. Во время ожидания поля и кнопка заблокированы; тайм-аут сетевого запроса равен 10 секундам.

В разработке работают Express на порту 3000 и Vite на порту 5173 с proxy /api. Для демонстрации выполняется сборка, после чего Express самостоятельно раздаёт содержимое dist и API на порту 3000. Относительный адрес API позволяет открывать сайт по IP компьютера без изменения клиентского кода.

Адаптивность реализована в styles.css: на широком экране поля располагаются в две колонки, при ширине до 600 px — в одну, при ширине до 360 px уменьшаются отступы. Подписи, фокус и области объявления ошибок и результата повышают доступность формы.

## 2.3.5. lab-01/src/App.jsx

Форма диапазона и состояния интерфейса. Проверяет ввод, асинхронно запрашивает результат, блокирует повторную отправку и отображает ошибки.

```
import { useRef, useState } from 'react';
import { requestRandom } from './api.js';

const LIMIT = 1_000_000_000;

export default function App() {

  const [min, setMin] = useState('1');
  const [max, setMax] = useState('100');

  const [pending, setPending] = useState(false);
  const [result, setResult] = useState(null);
  const [error, setError] = useState('');

  const inFlight = useRef(false);

  function changeValue(setValue, value) {
    setValue(value);
    setResult(null);
    setError('');
  }

  async function handleSubmit(event) {

    event.preventDefault();
    if (inFlight.current) return;
    setError('');
    setResult(null);

    if (!min.trim() || !max.trim()) {
      setError('Заполните оба поля диапазона.');
      return;
    }

    const lower = Number(min);
    const upper = Number(max);
    if (!Number.isInteger(lower) || !Number.isInteger(upper)) {
      setError('Минимум и максимум должны быть целыми числами.');
      return;
    }

    if (Math.abs(lower) > LIMIT || Math.abs(upper) > LIMIT) {
      setError('Границы должны быть от −1 000 000 000 до 1 000 000 000.');
      return;
    }
    if (lower > upper) {
      setError('Минимум не должен превышать максимум.');
      return;
    }

    inFlight.current = true;
    setPending(true);
    try {

      setResult(await requestRandom(lower, upper));
    } catch (failure) {
      setError(failure instanceof Error ? failure.message : 'Не удалось получить результат.');
    } finally {

      inFlight.current = false;
      setPending(false);
    }
  }

  return (
    <main className="card">
      <p className="caption">Лабораторная работа № 1 · Вариант 6</p>
      <h1>Генератор случайных чисел</h1>
      <p className="intro">Задайте диапазон целых чисел. Обе границы включаются в результат.</p>

      <form onSubmit={handleSubmit} noValidate aria-busy={pending}>
        <p id="range-hint" className="caption">Допустимые границы: от −1 000 000 000 до 1 000 000 000.</p>

        <fieldset disabled={pending}>
          <legend>Диапазон</legend>

          <div className="range-fields">
            <div className="field">

              <label htmlFor="min">Минимум</label>
              <input id="min" name="min" type="number" required step="1" min={-LIMIT} max={LIMIT}
                value={min} onChange={event => changeValue(setMin, event.target.value)}
                aria-describedby={error ? 'range-hint form-error' : 'range-hint'} />
            </div>
            <div className="field">
              <label htmlFor="max">Максимум</label>
              <input id="max" name="max" type="number" required step="1" min={-LIMIT} max={LIMIT}
                value={max} onChange={event => changeValue(setMax, event.target.value)}
                aria-describedby={error ? 'range-hint form-error' : 'range-hint'} />
            </div>
          </div>

          <button type="submit">{pending ? 'Генерация…' : 'Сгенерировать'}</button>
        </fieldset>
      </form>

      <p id="form-error" role="alert" className="error" aria-atomic="true">{error}</p>

      <section className="result" aria-live="polite" aria-atomic="true" aria-label="Результат генерации">
        {pending ? <p>Ожидаем ответ сервера…</p> : result ? <>
          <p className="result-label">Случайное число</p>
          <output>{result.value}</output>
          <p>Диапазон: от {result.min} до {result.max}</p>
        </> : <p>Введите диапазон и нажмите «Сгенерировать».</p>}
      </section>
    </main>
  );
}
```

## 2.3.6. lab-01/src/api.js

Сетевой слой клиента: POST /api/random через fetch, отмена по тайм-ауту, проверка HTTP-статуса и структуры JSON-ответа.

```
const TIMEOUT_MS = 10_000;

export async function requestRandom(min, max) {

  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), TIMEOUT_MS);

  try {
    let response;
    try {

      response = await fetch('/api/random', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ min, max }),
        signal: controller.signal,
      });
    } catch (error) {

      if (controller.signal.aborted) throw error;
      throw new Error('Не удалось связаться с сервером. Проверьте подключение и повторите попытку.');
    }

    let data;
    try {

      data = await response.json();
    } catch (error) {
      if (controller.signal.aborted) throw error;
      throw new Error('Сервер вернул некорректный ответ. Попробуйте ещё раз.');
    }

    if (!response.ok) {
      const message = data?.error?.message;
      throw new Error(typeof message === 'string' && message.trim()
        ? message
        : `Ошибка сервера (HTTP ${response.status}). Попробуйте ещё раз.`);
    }

    if (data?.min !== min || data?.max !== max || !Number.isInteger(data?.value)
      || data.value < min || data.value > max) {
      throw new Error('Сервер вернул некорректный результат. Попробуйте ещё раз.');
    }
    return data;
  } catch (error) {
    if (controller.signal.aborted) {
      throw new Error('Сервер не ответил за 10 секунд. Попробуйте ещё раз.');
    }
    throw error;
  } finally {

    clearTimeout(timeout);
  }
}
```

## 2.3.8. lab-01/server/random.js

Серверная логика: строгая проверка целых границ и генерация числа через crypto.randomInt с включением верхней границы.

```
import { randomInt } from 'node:crypto';

export const MIN_BOUND = -1_000_000_000;
export const MAX_BOUND = 1_000_000_000;

export class InvalidRangeError extends Error { }

export function generateRandom(input) {

  if (input === null || typeof input !== 'object' || Array.isArray(input)) {
    throw new InvalidRangeError('Тело запроса должно быть объектом с полями min и max');
  }

  const { min, max } = input;
  if (!Number.isInteger(min) || !Number.isInteger(max)) {
    throw new InvalidRangeError('Минимум и максимум должны быть целыми числами');
  }
  if (min < MIN_BOUND || min > MAX_BOUND || max < MIN_BOUND || max > MAX_BOUND) {
    throw new InvalidRangeError('Границы должны быть от −1 000 000 000 до 1 000 000 000');
  }

  if (min > max) {
    throw new InvalidRangeError('Минимум не должен превышать максимум');
  }

  const value = min === max ? min : randomInt(min, max + 1);
  return { min, max, value };
}
```

## 2.3.9. lab-01/server/app.js

Создаёт Express-приложение, маршруты диагностики и генерации, обработку ошибок, ограничение тела запроса и раздачу готовой сборки.

```
import express from 'express';
import { fileURLToPath } from 'node:url';
import { generateRandom, InvalidRangeError } from './random.js';

export const app = express();

app.get('/api/health', (_request, response) => {
  response.json({ status: 'ok' });
});

app.post('/api/random', (request, response, next) => {
  if (!request.is('application/json')) {

    return response.status(415).json({
      error: { code: 'UNSUPPORTED_MEDIA_TYPE', message: 'Ожидается Content-Type: application/json' },
    });
  }
  next();

}, express.json({ limit: '1kb', strict: false }), (request, response) => {
  response.set('Cache-Control', 'no-store');

  response.json(generateRandom(request.body));
});

app.use('/api', (_request, response) => {
  response.status(404).json({
    error: { code: 'NOT_FOUND', message: 'API-маршрут не найден' },
  });
});

app.use(express.static(fileURLToPath(new URL('../dist/', import.meta.url))));

app.use((error, _request, response, _next) => {

  let status = 500;
  let code = 'INTERNAL_ERROR';
  let message = 'Внутренняя ошибка сервера';

  if (error instanceof InvalidRangeError) {
    status = 400;
    code = 'INVALID_RANGE';
    message = error.message;
  } else if (error.type === 'entity.parse.failed') {
    status = 400;
    code = 'INVALID_JSON';
    message = 'Некорректный JSON';
  } else if (error.type === 'entity.too.large') {
    status = 413;
    code = 'PAYLOAD_TOO_LARGE';
    message = 'Размер тела запроса не должен превышать 1 КБ';
  } else if (error.status === 415) {
    status = 415;
    code = 'UNSUPPORTED_MEDIA_TYPE';
    message = 'Неподдерживаемая кодировка запроса';
  } else {
    console.error(error);
  }

  response.status(status).json({ error: { code, message } });
});
```

Критерии проверки: для 1…100 возвращается целое число внутри диапазона; для −5…−5 — число −5; обратный диапазон возвращает HTTP 400. В браузере проверяются отправка кнопкой и Enter, ошибки ввода, состояние ожидания и компоновка на узком экране. В tests находятся проверки генератора, HTTP API и клиентского сетевого слоя.

Доступность API с телефона была подтверждена. При проверке веб-интерфейса на телефоне ранее наблюдалась белая страница; успешное устранение этой проблемы в материалах работы не зафиксировано. Поэтому полная демонстрация веб-интерфейса на Авроре требует отдельного подтверждения.
