# 2. Выполнение

## 2.1. Архитектура и реализация

Браузерный UI в public/app.js собирает два числа и операцию, показывает состояние и ошибки; асинхронный HTTP-сервис в public/api.js отправляет JSON через XMLHttpRequest. Чистая функция server/calculator.js выполняет вычисления и валидацию; Express-слой server/app.js отвечает за маршруты и HTTP-статусы. CSS содержит @media для узких экранов. Сервер не доверяет проверке формы.

Flutter-клиент разделён на models/calculation.dart, services/calculator_service.dart, controllers/calculator_controller.dart и ui/calculator_page.dart. Сервис отвечает за WebSocket/JSON и закрытие ресурсов, контроллер — за состояние, тайм-аут и соответствие request_id, UI — за виджеты. Python app/calculator.py вычисляет, protocol.py разбирает JSON, main.py обслуживает соединение. Ошибка деления на ноль не разрывает прикладной сеанс.

Операции используют обычные double, а не десятичную финансовую арифметику. Ограничение операндов ±10¹² не разрешает NaN/Infinity. История хранит 10 записей только в памяти. WebSocket/Flutter-код впоследствии проверен после разрешения пользователя; нативный сценарий с реальным сервером прошёл. Ограничения тестового runner приведены в docs/VALIDATION.md.

## 2.3.2. lab1/lab-01/public/api.js

HTTP-транспорт или обработка действий пользователя.

```
var CalculatorApi = (function () {
  'use strict';

  function request(data, done) {
    var xhr = new XMLHttpRequest();
    xhr.open('POST', '/api/calculate'); xhr.timeout = 10000;
    xhr.setRequestHeader('Content-Type', 'application/json');
    xhr.onload = function () {
      try {
        var body = JSON.parse(xhr.responseText);
        if (xhr.status !== 200) throw new Error(body.error.message);
        if (body.a !== data.a || body.b !== data.b || body.operation !== data.operation || typeof body.value !== 'number' || !isFinite(body.value)) throw new Error('Неверный ответ сервера');
        done(null, body);
      } catch (error) { done(error); }
    };
    xhr.onerror = function () { done(new Error('Нет связи с сервером')); };
    xhr.ontimeout = function () { done(new Error('Сервер не ответил за 10 секунд')); };
    xhr.send(JSON.stringify(data));
  }
  return {request: request};
}());
```

## 2.3.7. lab1/lab-01/server/calculator.js

Арифметическая логика HTTP-калькулятора.

```
export function calculate(input) {
  if (!input || typeof input !== 'object' || Array.isArray(input)) throw new Error('Ожидается объект');
  const { a, b, operation } = input;
  if (Object.keys(input).some(key => !['a', 'b', 'operation'].includes(key))) throw new Error('Неизвестные поля');
  if (![a, b].every(value => typeof value === 'number' && Number.isFinite(value) && Math.abs(value) <= 1e12)) throw new Error('Введите два конечных числа от −10¹² до 10¹²');
  if (!['+', '-', '*', '/'].includes(operation)) throw new Error('Неизвестная операция');
  if (operation === '/' && b === 0) throw new Error('Деление на ноль запрещено');
  const value = operation === '+' ? a+b : operation === '-' ? a-b : operation === '*' ? a*b : a/b;
  if (!Number.isFinite(value)) throw new Error('Результат не является конечным числом');
  return { a, b, operation, value };
}
```

## 2.3.10. lab1/lab-02/backend/app/calculator.py

Чистая арифметическая логика с проверкой деления на ноль.

```
import math

LIMIT = 1_000_000_000_000

def calculate(a, b, operation):

    if type(a) not in (int, float) or type(b) not in (int, float):
        raise ValueError('Операнды должны быть числами')
    if not all(abs(x) <= LIMIT and math.isfinite(x) for x in (a, b)):
        raise ValueError('Операнды должны быть конечными числами от −10¹² до 10¹²')
    if operation not in ('+', '-', '*', '/'):
        raise ValueError('Допустимые операции: +, -, *, /')
    if operation == '/' and b == 0:
        raise ValueError('Деление на ноль запрещено')
    value = {'+': lambda: a+b, '-': lambda: a-b, '*': lambda: a*b, '/': lambda: a/b}[operation]()
    if not math.isfinite(value):
        raise ValueError('Результат не является конечным числом')
    return {'a': a, 'b': b, 'operation': operation, 'value': value}
```

import logging

from fastapi import FastAPI, WebSocket, WebSocketDisconnect

from .protocol import error, handle_message

logger = logging.getLogger(__name__)
app = FastAPI(title="ЛР1 · вариант 1 · Калькулятор", version="1.0.0")

@app.get("/health")
async def health():

    return {"status": "ok"}

@app.websocket("/ws/calculate")
async def calculator_socket(websocket: WebSocket):

    await websocket.accept()
    try:

        while True:

            event = await websocket.receive()
            if event["type"] == "websocket.disconnect":
                break
            if event.get("text") is None:
                await websocket.send_json(error("UNSUPPORTED_FRAME", "Отправляйте текстовый JSON, не бинарные данные"))
                continue
            try:

                response = handle_message(event["text"])
            except Exception:

                logger.exception("Ошибка обработки WebSocket-сообщения")
                await websocket.send_json(error("INTERNAL_ERROR", "Внутренняя ошибка сервера"))
                await websocket.close(code=1011)
                return

            await websocket.send_json(response)
    except WebSocketDisconnect:

        pass

## 2.3.12. lab1/lab-02/backend/app/protocol.py

Разбор WebSocket-запроса и формирование ответа с request_id.

```
import json
from .calculator import calculate

def error(code, message, request_id=None):
    return {'type': 'error', 'request_id': request_id, 'error': {'code': code, 'message': message}}

def reject_constant(value):
    raise ValueError('JSON не поддерживает ' + value)

def handle_message(text):
    if len(text.encode()) > 4096:
        return error('MESSAGE_TOO_LARGE', 'Сообщение больше 4096 байт')
    try:
        data = json.loads(text, parse_constant=reject_constant)
    except (ValueError, RecursionError):
        return error('INVALID_JSON', 'Некорректный JSON')
    if not isinstance(data, dict):
        return error('INVALID_REQUEST', 'Ожидается JSON-объект')
    request_id = data.get('request_id')
    if request_id is not None and (not isinstance(request_id, str) or not 1 <= len(request_id) <= 64):
        return error('INVALID_REQUEST', 'Некорректный request_id')
    if data.get('type') != 'calculate':
        return error('UNKNOWN_TYPE', 'Ожидается type: calculate', request_id)
    if set(data) - {'type', 'request_id', 'a', 'b', 'operation'}:
        return error('INVALID_REQUEST', 'Неизвестные поля', request_id)
    try:
        return {'type': 'result', 'request_id': request_id,
                **calculate(data.get('a'), data.get('b'), data.get('operation'))}
    except ValueError as exc:
        return error('INVALID_OPERATION', str(exc), request_id)
```

## 2.3.19. lab1/lab-02/calculator/lib/controllers/calculator_controller.dart

Контроллер Flutter: ввод, состояние, request_id и история.

```
import 'dart:async';
import 'package:flutter/foundation.dart';
import '../models/calculation.dart';
import '../services/calculator_service.dart';

class CalculatorController extends ChangeNotifier {
  CalculatorController(this.service) {
    _subscription = service.messages.listen(_receive, onError: (Object e) => _fail(e.toString()));
  }
  final CalculatorService service;
  late final StreamSubscription<Map<String, dynamic>> _subscription;
  final List<Calculation> _history = [];
  List<Calculation> get history => List.unmodifiable(_history);
  bool connected = false, connecting = false;
  bool get waiting => _pending != null;
  String? error;
  Map<String, dynamic>? _pending;
  Timer? _timer;
  int _sequence = 0, _generation = 0;
  bool _disposed = false;

  Future<void> connect(String address) async {
    final uri = Uri.tryParse(address.trim());
    if (uri == null || !['ws', 'wss'].contains(uri.scheme) || uri.host.isEmpty || uri.userInfo.isNotEmpty || uri.hasFragment) {
      error = 'Введите ws://<IP-ПК>:8100/ws/calculate'; notifyListeners(); return;
    }
    disconnect(); final generation = _generation;
    connecting = true; notifyListeners();
    try {
      await service.connect(uri);
      if (_disposed || generation != _generation) return;
      connecting = false; connected = true; notifyListeners();
    } catch (e) { if (!_disposed && generation == _generation) _fail('Ошибка подключения: $e'); }
  }
  void calculate(String first, String second, String operation) {
    if (!connected || waiting) return;
    final a = double.tryParse(first.trim().replaceAll(',', '.'));
    final b = double.tryParse(second.trim().replaceAll(',', '.'));
    if (a == null || b == null || !a.isFinite || !b.isFinite || a.abs() > 1e12 || b.abs() > 1e12 || !['+', '-', '*', '/'].contains(operation)) {
      error = 'Введите два конечных числа от −10¹² до 10¹²'; notifyListeners(); return;
    }
    error = null;
    _pending = {'type': 'calculate', 'request_id': 'calc-${++_sequence}', 'a': a, 'b': b, 'operation': operation};
    _timer = Timer(const Duration(seconds: 10), () => _fail('Сервер не ответил за 10 секунд'));
    try { service.send(_pending!); } catch (e) { _fail(e.toString()); return; }
    notifyListeners();
  }
  void _receive(Map<String, dynamic> data) {
    if (_disposed || _pending == null || data['request_id'] != _pending!['request_id']) return;
    try {
      if (data['type'] == 'error') {
        error = (data['error'] as Map<String, dynamic>)['message'] as String;
      } else if (data['type'] == 'result') {
        final result = Calculation.fromJson(data);
        if (result.a != _pending!['a'] || result.b != _pending!['b'] || result.operation != _pending!['operation']) {
          throw const FormatException('Ответ не соответствует запросу');
        }
        _history.insert(0, result);
        if (_history.length > 10) _history.removeLast();
      } else { throw const FormatException('Неизвестный тип ответа'); }
      _timer?.cancel(); _pending = null; notifyListeners();
    } catch (e) { _fail(e.toString()); }
  }
  void _fail(String message) { if (_disposed) return; disconnect(); error = message; notifyListeners(); }
  void disconnect() {
    ++_generation; service.disconnect(); _timer?.cancel(); _pending = null;
    connecting = false; connected = false; error = null;
    if (!_disposed) notifyListeners();
  }
  @override
  void dispose() {
    _disposed = true; disconnect(); unawaited(_subscription.cancel()); service.dispose(); super.dispose();
  }
}

class Calculation {
  const Calculation(this.a, this.b, this.operation, this.value);
  final double a, b, value;
  final String operation;
  factory Calculation.fromJson(Map<String, dynamic> data) {
    if (data['a'] is! num || data['b'] is! num || data['value'] is! num ||
        !['+', '-', '*', '/'].contains(data['operation'])) {
      throw const FormatException('Неверный ответ сервера');
    }
    final result = Calculation((data['a'] as num).toDouble(), (data['b'] as num).toDouble(),
        data['operation'] as String, (data['value'] as num).toDouble());
    if (![result.a, result.b, result.value].every((n) => n.isFinite)) {
      throw const FormatException('Сервер вернул бесконечное число');
    }
    return result;
  }
  @override
  String toString() => '$a $operation $b = $value';
}

## 2.3.22. lab1/lab-02/calculator/lib/services/calculator_service.dart

WebSocket-транспорт Flutter и обработка JSON-кадров.

```
import 'dart:async';
import 'dart:convert';
import 'dart:io';

abstract class CalculatorService {
  Stream<Map<String, dynamic>> get messages;
  Future<void> connect(Uri uri);
  void send(Map<String, dynamic> request);
  void disconnect();
  void dispose();
}

class WebSocketCalculatorService implements CalculatorService {
  final _events = StreamController<Map<String, dynamic>>.broadcast();
  WebSocket? _socket;
  StreamSubscription<dynamic>? _subscription;
  int _generation = 0;
  @override
  Stream<Map<String, dynamic>> get messages => _events.stream;
  @override
  Future<void> connect(Uri uri) async {
    disconnect();
    final generation = _generation;
    final socket = await WebSocket.connect(uri.toString()).timeout(const Duration(seconds: 10),
      onTimeout: () => throw TimeoutException('Не удалось подключиться'));
    if (generation != _generation) { unawaited(socket.close()); return; }
    _socket = socket;
    _subscription = socket.listen((dynamic frame) {
      try {
        if (frame is! String) throw const FormatException('Ожидается текстовый JSON');
        _events.add(jsonDecode(frame) as Map<String, dynamic>);
      } catch (error) { _events.addError(error); }
    }, onError: (Object error) => _events.addError(error),
      onDone: () { if (generation == _generation) _events.addError(StateError('Соединение закрыто')); });
  }
  @override
  void send(Map<String, dynamic> request) {
    if (_socket == null) throw StateError('Нет соединения');
    _socket!.add(jsonEncode(request));
  }
  @override
  void disconnect() {
    ++_generation;
    unawaited(_subscription?.cancel()); _subscription = null;
    unawaited(_socket?.close()); _socket = null;
  }
  @override
  void dispose() { disconnect(); unawaited(_events.close()); }
}
